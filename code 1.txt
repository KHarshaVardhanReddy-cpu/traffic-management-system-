from flask import Flask, jsonify, request
from flask_cors import CORS
from traffic import TrafficManager

app = Flask(__name__)
CORS(app)

traffic = TrafficManager()

# Home
@app.route("/")
def home():
    return jsonify({
        "message": "Smart Traffic Management System Backend",
        "status": "Running"
    })


# Get traffic information
@app.route("/api/traffic", methods=["GET"])
def get_traffic():
    return jsonify(traffic.get_status())


# Update vehicle count
@app.route("/api/traffic", methods=["POST"])
def update_traffic():
    data = request.json

    road = data.get("road")
    vehicles = data.get("vehicles")

    if road not in [1, 2, 3, 4]:
        return jsonify({"error": "Road must be between 1 and 4"}), 400

    if not isinstance(vehicles, int) or vehicles < 0:
        return jsonify({"error": "Invalid vehicle count"}), 400

    traffic.update_vehicles(road, vehicles)

    return jsonify({
        "message": "Traffic updated successfully",
        "data": traffic.get_status()
    })


# Automatic traffic control
@app.route("/api/control", methods=["POST"])
def automatic_control():
    result = traffic.select_signal()

    return jsonify(result)


# Emergency vehicle
@app.route("/api/emergency", methods=["POST"])
def emergency():
    data = request.json
    road = data.get("road")

    if road not in [1, 2, 3, 4]:
        return jsonify({"error": "Invalid road"}), 400

    result = traffic.emergency_priority(road)

    return jsonify(result)


if __name__ == "__main__":
    app.run(debug=True, host="0.0.0.0", port=5000)



class TrafficManager:

    def __init__(self):
        self.vehicles = {
            1: 0,
            2: 0,
            3: 0,
            4: 0
        }

        self.signal = {
            1: "RED",
            2: "RED",
            3: "RED",
            4: "RED"
        }

    def update_vehicles(self, road, vehicles):
        self.vehicles[road] = vehicles

    def get_status(self):
        return {
            "vehicles": self.vehicles,
            "signal": self.signal
        }

    def select_signal(self):

        # Find road with maximum traffic
        road = max(self.vehicles, key=self.vehicles.get)

        # Make all signals RED
        for r in self.signal:
            self.signal[r] = "RED"

        # Give GREEN signal to highest traffic road
        self.signal[road] = "GREEN"

        return {
            "message": "Automatic signal selected",
            "green_road": road,
            "signal": self.signal
        }

    def emergency_priority(self, road):

        # Make all signals RED
        for r in self.signal:
            self.signal[r] = "RED"

        # Emergency road gets GREEN
        self.signal[road] = "GREEN"

        return {
            "message": "Emergency priority activated",
            "emergency_road": road,
            "signal": self.signal
        }

    road INT NOT NULL,
    vehicle_count INT NOT NULL,
    signal_status VARCHAR(10) NOT NULL,
    emergency BOOLEAN DEFAULT FALSE,
    recorded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
